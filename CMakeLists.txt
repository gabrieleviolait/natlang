cmake_minimum_required(VERSION 3.16)
project(NatLang VERSION 0.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/runtime.hpp" NAT_RUNTIME_SOURCE)
configure_file(src/embedded_runtime.in.hpp "${CMAKE_CURRENT_BINARY_DIR}/embedded_runtime.hpp" @ONLY)
add_executable(natc src/natc.cpp)
target_include_directories(natc PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(natc PRIVATE /W4 /EHsc /utf-8)
else()
    target_compile_options(natc PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS natc RUNTIME DESTINATION bin)

# Dependency-free native Windows GUI. natc CLI continues to build cross-platform.
if(WIN32)
    add_executable(natlang-studio WIN32 studio/main.cpp)
    target_compile_features(natlang-studio PRIVATE cxx_std_20)
    target_link_libraries(natlang-studio PRIVATE comdlg32 shell32 user32 gdi32)
    if(MSVC)
        target_compile_options(natlang-studio PRIVATE /W4 /EHsc /utf-8)
        set_property(TARGET natlang-studio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        set_property(TARGET natc PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    else()
        target_compile_options(natlang-studio PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    install(TARGETS natlang-studio RUNTIME DESTINATION bin)
endif()
